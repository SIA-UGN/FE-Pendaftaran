<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use App\Models\Payment;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class PaymentController extends Controller
{
    /**
     * Get payment information for current user
     * 
     * Endpoint: GET /api/payment/my
     * Headers: Authorization: Bearer {token}
     * 
     * Response:
     * {
     *   "status": "success",
     *   "data": {
     *     "payment": {
     *       "id": 1,
     *       "student_name": "Faradis Yulianto",
     *       "registration_number": "REG-2024-001",
     *       "amount": 300000,
     *       "status": "pending",
     *       "deadline": "2024-12-31T23:59:59Z",
     *       "bank_accounts": [...]
     *     }
     *   }
     * }
     */
    public function getMyPayment(Request $request)
    {
        try {
            $user = $request->user();
            $registration = $user->registration;

            if (!$registration) {
                return response()->json([
                    'status' => 'error',
                    'message' => 'Registration not found'
                ], 404);
            }

            // Cek apakah sudah selesai isi data (step 1-5)
            if ($registration->current_step < 5) {
                return response()->json([
                    'status' => 'error',
                    'message' => 'Please complete all registration steps first (step 1-5)'
                ], 403);
            }

            // Ambil atau buat payment record
            $payment = $registration->payment;
            
            if (!$payment) {
                // Auto-create payment saat pertama kali akses
                $payment = Payment::create([
                    'registration_id' => $registration->id,
                    'amount' => 300000, // Fixed amount Rp 300.000
                    'status' => 'pending',
                    'deadline' => now()->addDays(3), // 3 hari untuk bayar
                ]);

                Log::info('Payment record auto-created', [
                    'registration_id' => $registration->id,
                    'payment_id' => $payment->id
                ]);
            }

            // Placeholder bank accounts (hardcoded untuk sementara)
            $bankAccounts = [
                [
                    'id' => 1,
                    'bank_name' => 'Bank BCA',
                    'account_number' => '1234567890',
                    'account_holder' => 'Universitas Global Nusantara',
                ],
                [
                    'id' => 2,
                    'bank_name' => 'Bank BNI',
                    'account_number' => '0987654321',
                    'account_holder' => 'Universitas Global Nusantara',
                ],
                [
                    'id' => 3,
                    'bank_name' => 'Bank Mandiri',
                    'account_number' => '1122334455',
                    'account_holder' => 'Universitas Global Nusantara',
                ],
            ];

            return response()->json([
                'status' => 'success',
                'data' => [
                    'payment' => [
                        'id' => $payment->id,
                        'student_name' => $registration->profile->full_name ?? 'N/A',
                        'registration_number' => $registration->registration_number,
                        'amount' => (float) $payment->amount,
                        'status' => $payment->status,
                        'deadline' => $payment->deadline?->toIso8601String(),
                        'rejection_reason' => $payment->rejection_reason,
                        'payment_proof' => $payment->payment_proof,
                        'uploaded_at' => $payment->uploaded_at?->toIso8601String(),
                        'verified_at' => $payment->verified_at?->toIso8601String(),
                        'bank_accounts' => $bankAccounts,
                    ]
                ]
            ], 200);

        } catch (\Exception $e) {
            Log::error('Failed to get payment info', [
                'error' => $e->getMessage(),
                'trace' => $e->getTraceAsString()
            ]);

            return response()->json([
                'status' => 'error',
                'message' => 'Failed to get payment info: ' . $e->getMessage()
            ], 500);
        }
    }

    /**
     * Upload payment proof
     * 
     * Endpoint: POST /api/payment/upload-proof
     * Headers: Authorization: Bearer {token}
     * Body: {
     *   "payment_proof": "/storage/documents/payment/123_proof.jpg"
     * }
     * 
     * Response:
     * {
     *   "status": "success",
     *   "message": "Payment proof uploaded successfully",
     *   "data": {
     *     "payment": {
     *       "id": 1,
     *       "status": "waiting_verification",
     *       "payment_proof": "/storage/documents/payment/123_proof.jpg",
     *       "uploaded_at": "2024-12-31T10:30:00Z"
     *     }
     *   }
     * }
     */
    public function uploadPaymentProof(Request $request)
    {
        try {
            // Validasi input
            $validated = $request->validate([
                'payment_proof' => 'required|string|max:500',
            ]);

            $user = $request->user();
            $registration = $user->registration;

            if (!$registration) {
                return response()->json([
                    'status' => 'error',
                    'message' => 'Registration not found'
                ], 404);
            }

            $payment = $registration->payment;

            if (!$payment) {
                return response()->json([
                    'status' => 'error',
                    'message' => 'Payment record not found. Please refresh the page.'
                ], 404);
            }

            // Cek apakah sudah expired
            if ($payment->deadline && now()->isAfter($payment->deadline)) {
                $payment->update(['status' => 'expired']);
                
                return response()->json([
                    'status' => 'error',
                    'message' => 'Payment deadline has expired'
                ], 403);
            }

            // Cek apakah status memungkinkan untuk upload
            if (!in_array($payment->status, ['pending', 'rejected'])) {
                return response()->json([
                    'status' => 'error',
                    'message' => 'Payment status does not allow upload. Current status: ' . $payment->status
                ], 403);
            }

            DB::beginTransaction();

            try {
                // Update payment record dengan URL yang sudah di-upload
                $payment->update([
                    'payment_proof' => $validated['payment_proof'],
                    'status' => 'waiting_verification',
                    'uploaded_at' => now(),
                ]);

                // Update registration status
                $registration->update([
                    'status' => 'waiting_payment_verification',
                    'current_step' => 6,
                ]);

                DB::commit();

                Log::info('Payment proof uploaded', [
                    'payment_id' => $payment->id,
                    'registration_id' => $registration->id,
                    'proof_url' => $validated['payment_proof']
                ]);

                return response()->json([
                    'status' => 'success',
                    'message' => 'Payment proof uploaded successfully',
                    'data' => [
                        'payment' => [
                            'id' => $payment->id,
                            'status' => $payment->status,
                            'payment_proof' => $payment->payment_proof,
                            'uploaded_at' => $payment->uploaded_at->toIso8601String(),
                        ]
                    ]
                ], 200);

            } catch (\Exception $e) {
                DB::rollBack();
                throw $e;
            }

        } catch (\Illuminate\Validation\ValidationException $e) {
            return response()->json([
                'status' => 'error',
                'message' => 'Validation failed',
                'errors' => $e->errors()
            ], 422);

        } catch (\Exception $e) {
            Log::error('Failed to upload payment proof', [
                'error' => $e->getMessage(),
                'trace' => $e->getTraceAsString()
            ]);

            return response()->json([
                'status' => 'error',
                'message' => 'Failed to upload payment proof: ' . $e->getMessage()
            ], 500);
        }
    }

    /**
     * Get payment status only (optional endpoint)
     * 
     * Endpoint: GET /api/payment/status
     * Headers: Authorization: Bearer {token}
     * 
     * Response:
     * {
     *   "status": "success",
     *   "data": {
     *     "status": "waiting_verification",
     *     "uploaded_at": "2024-12-31T10:30:00Z"
     *   }
     * }
     */
    public function getPaymentStatus(Request $request)
    {
        try {
            $user = $request->user();
            $registration = $user->registration;
            $payment = $registration?->payment;

            if (!$payment) {
                return response()->json([
                    'status' => 'success',
                    'data' => [
                        'status' => 'pending',
                        'uploaded_at' => null,
                        'verified_at' => null,
                    ]
                ], 200);
            }

            return response()->json([
                'status' => 'success',
                'data' => [
                    'status' => $payment->status,
                    'uploaded_at' => $payment->uploaded_at?->toIso8601String(),
                    'verified_at' => $payment->verified_at?->toIso8601String(),
                ]
            ], 200);

        } catch (\Exception $e) {
            Log::error('Failed to get payment status', [
                'error' => $e->getMessage()
            ]);

            return response()->json([
                'status' => 'error',
                'message' => 'Failed to get payment status'
            ], 500);
        }
    }

    /**
     * Get payment info (alias untuk getMyPayment)
     * 
     * Endpoint: GET /api/payment/info
     */
    public function getPaymentInfo(Request $request)
    {
        return $this->getMyPayment($request);
    }
}

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Payment extends Model
{
    use HasFactory;

    /**
     * The table associated with the model.
     *
     * @var string
     */
    protected $table = 'payments';

    /**
     * The attributes that are mass assignable.
     *
     * @var array<int, string>
     */
    protected $fillable = [
        'registration_id',
        'amount',
        'status',
        'payment_proof',
        'rejection_reason',
        'deadline',
        'uploaded_at',
        'verified_at',
        'verified_by',
    ];

    /**
     * The attributes that should be cast.
     *
     * @var array<string, string>
     */
    protected $casts = [
        'amount' => 'decimal:2',
        'deadline' => 'datetime',
        'uploaded_at' => 'datetime',
        'verified_at' => 'datetime',
    ];

    /**
     * Status constants
     */
    const STATUS_PENDING = 'pending';
    const STATUS_WAITING_VERIFICATION = 'waiting_verification';
    const STATUS_VERIFIED = 'verified';
    const STATUS_REJECTED = 'rejected';
    const STATUS_EXPIRED = 'expired';

    /**
     * Get all available statuses
     *
     * @return array
     */
    public static function getStatuses()
    {
        return [
            self::STATUS_PENDING,
            self::STATUS_WAITING_VERIFICATION,
            self::STATUS_VERIFIED,
            self::STATUS_REJECTED,
            self::STATUS_EXPIRED,
        ];
    }

    /**
     * Relasi ke Registration (One-to-One)
     * Setiap payment milik satu registration
     */
    public function registration()
    {
        return $this->belongsTo(Registration::class);
    }

    /**
     * Relasi ke User yang melakukan verifikasi (One-to-One)
     * Admin yang verify payment
     */
    public function verifiedBy()
    {
        return $this->belongsTo(User::class, 'verified_by');
    }

    /**
     * Check if payment is expired
     *
     * @return bool
     */
    public function isExpired()
    {
        if (!$this->deadline) {
            return false;
        }

        return now()->isAfter($this->deadline);
    }

    /**
     * Check if payment can be uploaded (pending or rejected dan belum expired)
     *
     * @return bool
     */
    public function canUpload()
    {
        return in_array($this->status, [self::STATUS_PENDING, self::STATUS_REJECTED]) 
               && !$this->isExpired();
    }

    /**
     * Check if payment is pending
     *
     * @return bool
     */
    public function isPending()
    {
        return $this->status === self::STATUS_PENDING;
    }

    /**
     * Check if payment is waiting verification
     *
     * @return bool
     */
    public function isWaitingVerification()
    {
        return $this->status === self::STATUS_WAITING_VERIFICATION;
    }

    /**
     * Check if payment is verified
     *
     * @return bool
     */
    public function isVerified()
    {
        return $this->status === self::STATUS_VERIFIED;
    }

    /**
     * Check if payment is rejected
     *
     * @return bool
     */
    public function isRejected()
    {
        return $this->status === self::STATUS_REJECTED;
    }

    /**
     * Scope untuk filter by status
     */
    public function scopeByStatus($query, $status)
    {
        return $query->where('status', $status);
    }

    /**
     * Scope untuk payment yang expired
     */
    public function scopeExpired($query)
    {
        return $query->where('status', self::STATUS_PENDING)
                     ->where('deadline', '<', now());
    }

    /**
     * Scope untuk payment yang waiting verification
     */
    public function scopeWaitingVerification($query)
    {
        return $query->where('status', self::STATUS_WAITING_VERIFICATION);
    }
}

FILE 9: TROUBLESHOOTING
================================================================================

COMMON ISSUES & SOLUTIONS:

1. ERROR: "Class 'App\Models\Payment' not found"
   SOLUTION:
   - Pastikan Payment.php ada di app/Models/
   - Run: composer dump-autoload
   - Clear cache: php artisan config:clear

2. ERROR: "Table 'payments' doesn't exist"
   SOLUTION:
   - Check migration sudah run: php artisan migrate:status
   - Run migration: php artisan migrate
   - Check database connection di .env

3. ERROR: "Registration not found"
   SOLUTION:
   - User belum punya registration record
   - Pastikan user sudah isi data step 1-5
   - Check tabel registrations, pastikan ada record dengan user_id yang login

4. ERROR: "Please complete all registration steps first"
   SOLUTION:
   - current_step di tabel registrations masih < 5
   - User harus selesaikan step 1-5 dulu
   - Update current_step di database untuk testing:
     UPDATE registrations SET current_step = 5 WHERE user_id = {your_user_id};

5. ERROR: 500 Internal Server Error
   SOLUTION:
   - Check laravel.log: storage/logs/laravel.log
   - Enable debug di .env: APP_DEBUG=true
   - Check database connection
   - Check semua required fields ada

6. ERROR: "Column not found: status"
   SOLUTION:
   - Migration belum jalan
   - Atau nama kolom salah
   - Re-run migration: php artisan migrate:rollback && php artisan migrate

7. ERROR: Foreign key constraint fails
   SOLUTION:
   - Tabel registrations belum ada
   - Atau registration_id tidak valid
   - Check foreign key di migration

8. Frontend tidak menerima data bank_accounts

FLOW PAYMENT SYSTEM:

1. User selesai isi data step 1-5
   ↓
2. current_step = 5, status = 'pending'
   ↓
3. User klik menu "Pembayaran"
   ↓
4. Frontend hit: GET /api/payment/my
   ↓
5. Backend cek: apakah payment record sudah ada?
   - Tidak ada → Auto-create payment record (status: pending)
   - Sudah ada → Return existing payment
   ↓
6. Frontend tampilkan:
   - Ringkasan pembayaran (nama, nomor, amount)
   - Countdown timer (deadline)
   - Info rekening bank (3 bank)
   - Form upload bukti bayar
   ↓
7. User upload file di frontend
   ↓
8. Frontend upload file via POST /api/upload (existing endpoint)
   ↓
9. Get URL hasil upload: /storage/documents/payment/xxx.jpg
   ↓
10. Frontend hit: POST /api/payment/upload-proof
    Body: { "payment_proof": "/storage/documents/payment/xxx.jpg" }
    ↓
11. Backend:
    - Update payment.payment_proof
    - Update payment.status = 'waiting_verification'
    - Update registration.status = 'waiting_payment_verification'
    - Update registration.current_step = 6
    ↓
12. Frontend show success message
    ↓
13. User tunggu verifikasi admin (1x24 jam)
    ↓
14. Admin login → lihat list payment waiting verification
    ↓
15. Admin verify payment:
    - Approve → status = 'verified'
    - Reject → status = 'rejected' (user bisa upload ulang)
    ↓
16. User bisa cek status di halaman status pendaftaran
