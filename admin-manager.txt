     * tags={"3. Admin"},
     * summary="Create user baru (Admin/Manager)",
     * description="Hanya untuk role 'admin'.",
     * security={{"sanctum":{}}},
     * @OA\RequestBody(
     * required=true,
     * @OA\JsonContent(
     * required={"name", "email", "password", "password_confirmation", "role"},
     * @OA\Property(property="name", type="string", example="Manager Baru"),
     * @OA\Property(property="email", type="string", example="manager@example.com"),
     * @OA\Property(property="password", type="string", format="password", example="password123"),
     * @OA\Property(property="password_confirmation", type="string", format="password", example="password123"),
     * @OA\Property(property="role", type="string", example="manager", description="Hanya 'admin' atau 'manager'")
     * )
     * ),
     * @OA\Response(
     * response=201,
     * description="User berhasil dibuat",
     * @OA\JsonContent(
     * @OA\Property(property="success", type="boolean", example=true),
     * @OA\Property(property="data", ref="#/components/schemas/UserResource")
     * )
     * ),
     * @OA\Response(response=403, description="Akses ditolak", @OA\JsonContent(ref="#/components/schemas/ErrorForbidden")),
     * @OA\Response(response=401, description="Unauthenticated")
     * )
     */
    public function createUser(StoreUserRequest $request): JsonResponse
    {
        try {
            $user = auth('sanctum')->user();
            $data = $request->validated();

            // Check authorization: hanya admin
            if ($user->role !== 'admin') {
                return response()->json([
                    'success' => false,
                    'message' => 'Hanya admin yang bisa membuat user baru',
                ], 403);
            }

            // 📌 PERBAIKAN: Hapus username dari array create agar tidak menyebabkan Error 500
            $newUser = User::create([
                'name' => $request->name,
                'email' => $request->email,
                'username' => $data['username'] ?? null, // Simpan jika ada, jika tidak, simpan null
                'phone' => $data['phone'] ?? null,
                'password' => Hash::make($request->password),
                'role' => $request->role, // admin, manager
                'is_active' => true,
            ]);

            return response()->json([
                'success' => true,
                'message' => 'User berhasil dibuat',
                'data' => new UserResource($newUser),
            ], 201);
        } catch (\Exception $e) {
            // Error 500 masih bisa terjadi jika ada masalah di request atau database selain kolom username
            return response()->json([
                'success' => false,
                'message' => 'Gagal membuat user',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

    /**
     * Get Payment Detail by Applicant ID (Profile ID)
     * React
     */
    public function getApplicantPayment($id): JsonResponse
    {
        try {
            $user = auth()->user();
            if (!in_array($user->role, ['manager', 'admin'])) {
                return response()->json(['success' => false, 'message' => 'Akses ditolak.'], 403);
            }

            $profile = ApplicantProfile::find($id);
            if (!$profile) {
                return response()->json(['success' => false, 'message' => 'Pendaftar tidak ditemukan'], 404);
            }

            $payment = Payment::with('paymentMethod')
                ->where('id_user', $profile->id_user)
                ->latest()
                ->first();

            if (!$payment) {
                return response()->json(['success' => false, 'message' => 'Data pembayaran belum tersedia.'], 404);
            }

            $methodName = $payment->paymentMethod 
                ? ($payment->paymentMethod->bank_name . ' - ' . $payment->paymentMethod->account_number) 
                : 'Transfer Manual';

            return response()->json([
                'success' => true,
                'data' => [
                    'id' => $payment->id,
                    'payment_code' => $payment->payment_code,
                    'amount' => (float) $payment->amount,
                    'paid_amount' => (float) $payment->paid_amount,
                    'status' => $payment->status,
                    'payment_date' => $payment->created_at->format('d M Y, H:i'),
                    'payment_method_name' => $methodName,
                    'sender_bank' => $payment->sender_bank,
                    'sender_account_number' => $payment->sender_account_number,
                    'sender_account_holder' => $payment->sender_account_holder,
                    'payment_proof_file' => $payment->payment_proof_file,
                    'proof_url' => $payment->payment_proof_file ? url('storage/' . $payment->payment_proof_file) : null,
                    'rejection_reason' => $payment->rejection_reason,
                ]
            ], 200);

        } catch (\Exception $e) {
            return response()->json(['success' => false, 'message' => 'Gagal memuat pembayaran', 'error' => $e->getMessage()], 500);
        }
    }

    /**
     * Verify or Reject Payment
     */
    public function verifyPayment(Request $request, $id): JsonResponse
    {
        try {
            // Validasi Role
            $user = auth()->user();
            if (!in_array($user->role, ['manager', 'admin'])) {
                return response()->json(['success' => false, 'message' => 'Akses ditolak.'], 403);
            }

            // Validasi Input
            $request->validate([
                'status' => 'required|in:verified,rejected',
                'rejection_reason' => 'required_if:status,rejected'
            ]);

            $payment = Payment::findOrFail($id);

            // Update Status
            $payment->status = $request->status;
            
            if ($request->status == 'verified') {
                $payment->verified_at = now();
                $payment->verified_by = $user->id_user;
                $payment->rejection_reason = null;
            } else {
                $payment->rejected_by = $user->id_user;
                $payment->rejection_reason = $request->rejection_reason;
            }

            $payment->save();

            return response()->json([
                'success' => true,
                'message' => 'Status pembayaran berhasil diperbarui.',
                'data' => $payment
            ], 200);

        } catch (\Exception $e) {
            return response()->json(['success' => false, 'message' => 'Gagal verifikasi pembayaran', 'error' => $e->getMessage()], 500);
        }
    }

    /**
     * Secure Image Viewer
     */
    public function getPaymentFile($id)
    {
        $payment = Payment::findOrFail($id);

        if (!$payment->payment_proof_file || !Storage::disk('public')->exists($payment->payment_proof_file)) {
            abort(404, 'File not found');
        }

        return response()->file(storage_path('app/public/' . $payment->payment_proof_file));
    }

     * Serve Achievement Certificate File securely
     * GET /api/admin/achievement-file/{id}
     */
    public function serveAchievement($id)
    {
        // Pastikan model di-import atau gunakan full namespace
        $achievement = \App\Models\ApplicantAchievement::find($id);

        if (!$achievement) {
            return response()->json(['message' => 'Data prestasi tidak ditemukan'], 404);
        }

        // Cek apakah path file ada di database
        if (!$achievement->certificate_path) {
            return response()->json(['message' => 'File sertifikat belum diupload'], 404);
        }

        $path = storage_path('app/public/' . $achievement->certificate_path);

        if (!file_exists($path)) {
            return response()->json(['message' => 'File fisik tidak ditemukan di server'], 404);
        }

        return response()->file($path);
    }

/**
 * @OA\Get(
 * path="/api/admin/applicants",
 * tags={"Admin & Manager"},
 * summary="Get list of applicants (Manager & Admin)",
 * security={{"sanctum":{}}},
 * @OA\Parameter(
 *     name="status",
 *     in="query",
 *     description="Filter by status: draft, submitted, approved, rejected",
 *     required=false,
 *     @OA\Schema(type="string")
 * ),
 * @OA\Parameter(
 *     name="search",
 *     in="query",
 *     description="Search by name, email, or program",
 *     required=false,
 *     @OA\Schema(type="string")
 * ),
 * @OA\Response(
 *     response=200,
 *     description="List of applicants",
 *     @OA\JsonContent(
 *         @OA\Property(property="success", type="boolean", example=true),
 *         @OA\Property(property="data", type="array",
 *             @OA\Items(
 *                 @OA\Property(property="id_profile", type="integer", example=1),
 *                 @OA\Property(property="full_name", type="string", example="John Doe"),
 *                 @OA\Property(property="email", type="string", example="john@email.com"),
 *                 @OA\Property(property="program_name", type="string", example="Teknik Informatika"),
 *                 @OA\Property(property="registration_number", type="string", example="REG-2025-0001"),
 *                 @OA\Property(property="registration_status", type="string", example="submitted"),
 *                 @OA\Property(property="created_at", type="string", example="2025-01-15")
 *             )
 *         )
 *     )
 * ),
 * @OA\Response(response=401, description="Unauthorized"),
 * @OA\Response(response=403, description="Forbidden - Only Manager/Admin")
 * )
 */
    public function getApplicants(Request $request): JsonResponse
    {
        try {
            // Cek role user
            $user = auth()->user();
            if (!in_array($user->role, ['manager', 'admin'])) {
                return response()->json([
                    'success' => false,
                    'message' => 'Akses ditolak. Hanya Manager dan Admin yang dapat mengakses.',
                ], 403);
            }

            $query = ApplicantProfile::with(['user', 'program'])
                ->where('registration_status', '!=', 'draft'); // Hanya tampilkan yang sudah submit

            // Filter by status
            if ($request->has('status') && $request->status != '') {
                $query->where('registration_status', $request->status);
            }

            // Search by name, email, or program
            if ($request->has('search') && $request->search != '') {
                $search = $request->search;
                $query->where(function($q) use ($search) {
                    $q->where('full_name', 'like', "%{$search}%")
                    ->orWhere('registration_number', 'like', "%{$search}%")
                    ->orWhereHas('user', function($q) use ($search) {
                        $q->where('email', 'like', "%{$search}%");
                    })
                    ->orWhereHas('program', function($q) use ($search) {
                        $q->where('name_program', 'like', "%{$search}%");
                    });
                });
            }

            $applicants = $query->orderBy('created_at', 'desc')->get();

            // Transform data
            $data = $applicants->map(function($applicant) {
                return [
                    'id_profile' => $applicant->id_profile,
                    'id_user' => $applicant->id_user,
                    'full_name' => $applicant->full_name,
                    'email' => $applicant->user->email ?? null,
                    'program_name' => $applicant->program->name_program ?? null,
                    'registration_number' => $applicant->registration_number,
                    'registration_status' => $applicant->registration_status,
                    'created_at' => $applicant->created_at->format('d M Y'),
                    'phone_number' => $applicant->phone_number,
                ];
            });

            return response()->json([
                'success' => true,
                'data' => $data,
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'message' => 'Gagal mengambil data pendaftar.',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

/**
 * @OA\Put(
 * path="/api/admin/applicants/{id}/status",
 * tags={"Admin & Manager"},
 * summary="Update applicant status (approve/reject)",
 * security={{"sanctum":{}}},
 * @OA\Parameter(
 *     name="id",
 *     in="path",
 *     required=true,
 *     description="Applicant Profile ID",
 *     @OA\Schema(type="integer")
 * ),
 * @OA\RequestBody(
 *     required=true,
 *     @OA\JsonContent(
 *         required={"status"},
 *         @OA\Property(property="status", type="string", enum={"approved", "rejected"}, example="approved"),
 *         @OA\Property(property="notes", type="string", example="Semua dokumen lengkap dan valid")
 *     )
 * ),
 * @OA\Response(
 *     response=200,
 *     description="Status updated successfully"
 * ),
 * @OA\Response(response=403, description="Forbidden"),
 * @OA\Response(response=404, description="Applicant not found")
 * )
 */
public function updateApplicantStatus(Request $request, $id): JsonResponse
    {
        try {
            $user = auth()->user();
            if (!in_array($user->role, ['manager', 'admin'])) {
                return response()->json([
                    'success' => false,
                    'message' => 'Akses ditolak. Hanya Manager dan Admin yang dapat mengubah status.',
                ], 403);
            }

            $request->validate([
                'status' => 'required|in:approved,rejected',
                'notes' => 'nullable|string|max:500', // 🔥 Validasi notes
            ]);

            $applicant = ApplicantProfile::find($id);
            
            if (!$applicant) {
                return response()->json([
                    'success' => false,
                    'message' => 'Data pendaftar tidak ditemukan.',
                ], 404);
            }

            // 🔥 Update status DAN notes
            $applicant->registration_status = $request->status;
            
            // Kolom rejection_reason digunakan untuk menyimpan notes/alasan.
            if ($request->has('notes')) {
                $applicant->rejection_reason = $request->notes; 
            }
            
            $applicant->save();

            $statusText = $request->status === 'approved' ? 'disetujui' : 'ditolak';

            return response()->json([
                'success' => true,
                'message' => "Pendaftar berhasil {$statusText}.",
                'data' => [
                    'id_profile' => $applicant->id_profile,
                    'registration_status' => $applicant->registration_status,
                    'notes' => $request->notes,
                ],
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'message' => 'Gagal mengubah status pendaftar.',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

/**

 *     name="id",

 *     required=true,
 *     @OA\Schema(type="integer")
 * ),
 * @OA\Response(
 *     response=200,
 *     description="Documents list"
 * )
 * )
 */
public function getApplicantDocuments($id): JsonResponse
{
    try {
        $user = auth()->user();
        if (!in_array($user->role, ['manager', 'admin'])) {
            return response()->json([
                'success' => false,
                'message' => 'Akses ditolak.',
            ], 403);

        $applicant = ApplicantProfile::find($id);

            return response()->json([
                'success' => false,
                'message' => 'Data pendaftar tidak ditemukan.',
            ], 404);
        }

        $documents = $applicant->documents()
            ->with('documentType')
            ->get()
            ->map(function($doc) {
                return [

                    'uploaded_at' => $doc->created_at->format('d M Y'),
                ];
            });

        return response()->json([

public function serveDocument($id)
{
    $document = \App\Models\ApplicantDocument::find($id);

    if (!$document) {
        return response()->json(['message' => 'Dokumen tidak ditemukan'], 404);
    }

    $path = storage_path('app/public/' . $document->file_path);

    if (!file_exists($path)) {
        return response()->json(['message' => 'File fisik tidak ditemukan'], 404);
    }

    return response()->file($path);
}

     *                 @OA\Property(property="total_revenue", type="number", format="float", example=60000000)
     *             )
     *         )
     *     ),
     *     @OA\Response(response=401, description="Unauthenticated"),
     *     @OA\Response(response=403, description="Access denied")
     * )
     */
    public function dashboard(): JsonResponse
    {
        try {
            $user = auth()->user();
            if (!in_array($user->role, ['manager', 'admin'])) {
                return response()->json([
                    'success' => false,
                    'message' => 'Akses ditolak.',
                ], 403);
            }

            $topPrograms = Program::withCount([
                'applicantProfiles as count' => function ($query) {
                    $query->whereIn('registration_status', ['draft', 'submitted', 'reviewed', 'approved', 'rejected']);
                }
            ])
            ->orderByDesc('count')
            ->get()
            ->map(function ($program, $index) {
                return [
                    'program' => $program->name_program,
                    'code' => $program->code_program,
                    'count' => $program->count,
                ];
            });

            $now = Carbon::now();

            $revenueThisMonth = Payment::where('status', 'verified')
                ->whereMonth('created_at', $now->month)
                ->whereYear('created_at', $now->year)
                ->sum('paid_amount');

            $revenueLastMonth = Payment::where('status', 'verified')
                ->whereMonth('created_at', $now->subMonth()->month)
                ->whereYear('created_at', $now->year)
                ->sum('paid_amount');

            if ($revenueLastMonth > 0) {
                $growthPercentage = (
                    ($revenueThisMonth - $revenueLastMonth) / $revenueLastMonth
                ) * 100;
            } else {
                $growthPercentage = $revenueThisMonth > 0 ? 100 : 0;
            }

            $data = [
                'total_applicants' => ApplicantProfile::count(),
                'pending_applicants' => ApplicantProfile::where('registration_status', 'submitted')->count(),
                'approved_applicants' => ApplicantProfile::where('registration_status', 'approved')->count(),
                'rejected_applicants' => ApplicantProfile::where('registration_status', 'rejected')->count(),
                'total_payments' => Payment::count(),
                'verified_payments' => Payment::where('status', 'verified')->count(),
                'pending_payments' => Payment::whereIn('status', ['pending', 'waiting_verification'])->count(),
                'total_revenue' => Payment::where('status', 'verified')->sum('paid_amount'),
                'total_managers' => User::where('role', 'manager')->count(),
                'programs' => $topPrograms,
                'growth_percentage'  => round($growthPercentage, 2),
            ];

            return response()->json([
                'success' => true,
                'data' => $data,
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'message' => 'Gagal mengambil data dashboard.',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

                ->get()

                ->map(fn ($items) => $items->count())

                    ->get()
                    ->groupBy(fn ($item) => $item->program->name_program ?? 'Unknown')
                    ->map(fn ($items) => $items->count()),
            
                'by_gender' => ApplicantProfile::whereNotNull('gender')
                    ->get()
                    ->groupBy('gender')
                    ->map(fn ($items) => $items->count()),
            
                'by_graduation_status' => collect($graduationStatuses)
                    ->mapWithKeys(fn ($status) => [$status => $byGraduationRaw[$status] ?? 0]),
            ];

                'success' => true,
                'data' => $data,

            return response()->json([

     *         )
     *     ),
     *     @OA\Response(response=401, description="Unauthenticated"),
     *     @OA\Response(response=403, description="Access denied - Admin only")
     * )

    public function yearlyRevenue(): JsonResponse
    {
        try {
            $user = auth()->user();
            if ($user->role !== 'admin') {
                return response()->json([
                    'success' => false,
                    'message' => 'Akses ditolak. Hanya admin yang dapat mengakses statistik ini.',
                ], 403);
            }

            $year = date('Y');
            $monthlyRevenue = Payment::where('status', 'verified')
                ->whereYear('verified_at', $year)
                ->selectRaw('MONTH(verified_at) as month, SUM(paid_amount) as revenue')
                ->groupBy('month')
                ->pluck('revenue', 'month')
                ->toArray();

            // Fill missing months with 0
            $allMonths = [];
            for ($i = 1; $i <= 12; $i++) {
                $allMonths[$i] = $monthlyRevenue[$i] ?? 0;
            }

            return response()->json([
                'success' => true,
                'data' => [
                    'year' => $year,
                    'monthly_revenue' => $allMonths,
                ],
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'message' => 'Gagal mengambil data pendapatan tahunan.',
                'error' => $e->getMessage(),
            ], 500);
        }

     *     security={{"sanctum":{}}},
     *     @OA\Parameter(
     *         name="id",
     *         in="path",
     *         required=true,
     *         description="Manager ID",
     *         @OA\Schema(type="integer")
     *     ),
     *     @OA\Response(
     *         response=200,
     *         description="Manager deleted successfully",
     *         @OA\JsonContent(
     *             @OA\Property(property="success", type="boolean", example=true),
     *             @OA\Property(property="message", type="string", example="Manager berhasil dihapus")
     *         )
     *     ),
     *     @OA\Response(response=401, description="Unauthenticated"),
     *     @OA\Response(response=403, description="Access denied - Admin only"),
     *     @OA\Response(response=404, description="Manager not found")
     * )
     */
    public function deleteManager($id): JsonResponse
    {
        try {
            $user = auth()->user();
            if ($user->role !== 'admin') {
                return response()->json([
                    'success' => false,
                    'message' => 'Akses ditolak. Hanya admin yang dapat menghapus manager.',
                ], 403);
            }

            $manager = User::where('id_user', $id)->where('role', 'manager')->first();
            
            if (!$manager) {
                return response()->json([
                    'success' => false,
                    'message' => 'Manager tidak ditemukan.',
                ], 404);
            }

            // Prevent deleting yourself
            if ($manager->id_user === $user->id_user) {
                return response()->json([
                    'success' => false,
                    'message' => 'Anda tidak dapat menghapus akun Anda sendiri.',
                ], 403);
            }

            $manager->delete();

            return response()->json([
                'success' => true,
                'message' => 'Manager berhasil dihapus.',
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'message' => 'Gagal menghapus manager.',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

public function storeProfile(StoreProfileRequest $request): JsonResponse
{
    try {
        $user = auth('sanctum')->user();
        $data = $request->validated();
        
        $data['id_user'] = $user->id_user; 
        
        $isExistingProfile = ApplicantProfile::where('id_user', $user->id_user)->exists();

        if (!$isExistingProfile) {
            
            if (empty($data['registration_number'])) {
                $data['registration_number'] = $this->registrationService->generateRegistrationNumber(); 
            }

            $data['graduation_status'] = 'Belum Lulus';
        }
        
        if ($isExistingProfile && isset($data['graduation_status'])) {
            unset($data['graduation_status']);
        }
        
        if (isset($data['email'])) {
            $user->email = $data['email'];
            $user->save();
            unset($data['email']); 
        }
        
        $profile = ApplicantProfile::updateOrCreate(
            ['id_user' => $user->id_user],
            $data
        );

        return response()->json([
            'success' => true,
            'message' => 'Profil berhasil disimpan',
            'data' => new ApplicantProfileResource($profile),
        ], 200);

    } catch (\Illuminate\Database\QueryException $e) {
        \Log::error('Store Profile DB Error (NOT NULL): ' . $e->getMessage());
        
        return response()->json([
            'success' => false,
            'message' => 'Gagal menyimpan profil: Ada field wajib database yang kosong.',
            'error_detail' => $e->getMessage(),
        ], 500);
    }
}

     * @OA\Property(property="data", ref="#/components/schemas/ApplicantDocumentResource")
     * )
     * ),
     * @OA\Response(response=422, description="Validasi gagal", @OA\JsonContent(ref="#/components/schemas/ErrorValidation")),
     * @OA\Response(response=401, description="Unauthenticated")
     * )
     */
    public function uploadDocument(UploadDocumentRequest $request): JsonResponse
{
    try {
        $user = auth('sanctum')->user();
        $profile = ApplicantProfile::where('id_user', $user->id_user)->first();

        if (!$profile) {
            return response()->json([
                'success' => false,
                'message' => 'Gagal mengupload dokumen. Data identitas (profil) harus diisi dan disimpan terlebih dahulu.',
                'error' => 'ApplicantProfile not found for this user.'
            ], 422);
        }
        
        // ✅ TAMBAHKAN: Cek dokumen lama dan hapus file-nya
        $existingDocument = ApplicantDocument::where('id_profile', $profile->id_profile)
            ->where('id_document_type', $request->id_document_type)
            ->first();
        
        if ($existingDocument && $existingDocument->file_path) {
            Storage::disk('public')->delete($existingDocument->file_path);
        }
        
        // Upload file baru
        $filePath = $request->file('file')->store('documents/' . $profile->id_profile, 'public');

        $document = ApplicantDocument::updateOrCreate(
            [
                'id_profile' => $profile->id_profile,
                'id_document_type' => $request->id_document_type,
            ],
            [
                'file_path' => $filePath,
                'verification_status' => 'pending',
                'upload_date' => now(),
            ]
        );

        return response()->json([
            'success' => true,
            'message' => 'Dokumen berhasil diupload',
            'data' => new ApplicantDocumentResource($document),
        ], 201);
    } catch (\Exception $e) {
        \Log::error('Upload Document Error: ' . $e->getMessage());
        return response()->json([
            'success' => false,
            'message' => 'Gagal mengupload dokumen',
            'error' => $e->getMessage(),
        ], 500);
    }
}

     * ),
     * @OA\Response(response=200, description="Data wali berhasil dihapus", @OA\JsonContent(ref="#/components/schemas/SuccessMessage")),
     * @OA\Response(response=403, description="Tidak berhak", @OA\JsonContent(ref="#/components/schemas/ErrorForbidden")),
     * @OA\Response(response=401, description="Unauthenticated")
     * )
     */
    public function deleteGuardian($id): JsonResponse
    {
        try {
            $user = auth('sanctum')->user();
            $guardian = Guardian::findOrFail($id);
            $profile = ApplicantProfile::where('id_user', $user->id_user)->firstOrFail();

            if ($guardian->id_profile !== $profile->id_profile) {
                return response()->json([
                    'success' => false,
                    'message' => 'Anda tidak berhak menghapus data wali ini',
                ], 403);
            }

            $guardian->delete();

            return response()->json([
                'success' => true,
                'message' => 'Data wali berhasil dihapus',
            ], 200);
        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'message' => 'Gagal menghapus data wali',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

    /**
     * @OA\Get(
     * path="/api/registration/status",
     * tags={"2. Pendaftaran"},
     * summary="Get status pendaftaran",
     * security={{"sanctum":{}}},
     * @OA\Response(
     * response=200,
     * description="Status pendaftaran berhasil diambil",
     * @OA\JsonContent(
     * @OA\Property(property="success", type="boolean", example=true),
     * @OA\Property(property="data", type="object",
     * @OA\Property(property="registration_number", type="string", example="REG-2025-0001"),
     * @OA\Property(property="status", type="string", example="submitted"),
     * @OA\Property(property="full_name", type="string", example="Nindya Pendaftar")
     * )
     * )
     * ),
     * @OA\Response(response=404, description="Profil tidak ditemukan", @OA\JsonContent(ref="#/components/schemas/ErrorNotFound")),
     * @OA\Response(response=401, description="Unauthenticated")
     * )
     */
    public function getRegistrationStatus(): JsonResponse
    {
        $user = auth('sanctum')->user();
        $profile = ApplicantProfile::where('id_user', $user->id_user)->first();

        if (!$profile) {
            return response()->json([
                'success' => true,
                'data' => [
                    'progress' => [
                        'profile' => false,
                        'guardians' => false,
                        'documents' => false,
                        'achievements' => false,
                        'submitted' => false,
                    ],
                    'profile' => null,
                    'guardians_count' => 0,
                    'documents_count' => 0,
                    'achievements_count' => 0,
                ]
            ], 200);
        }

        $guardiansCount = $profile->guardians()->count();
        $documentsCount = $profile->documents()->count();
        $achievementsCount = $profile->achievements()->count();

        return response()->json([
            'success' => true,
            'data' => [
                'progress' => [
                    'profile' => !empty($profile->full_name),
                    'guardians' => $guardiansCount > 0,
                    'documents' => $documentsCount > 0,
                    'achievements' => $achievementsCount > 0,
                    'submitted' => $profile->registration_status === 'submitted',
                ],
                'profile' => $profile,
                'guardians_count' => $guardiansCount,
                'documents_count' => $documentsCount,
                'achievements_count' => $achievementsCount,
            ]
        ], 200);
    }

    public function getAchievements(): JsonResponse
    {
        try {
            $user = Auth::user();
            $profile = ApplicantProfile::where('id_user', $user->id_user)->firstOrFail();

            // Ambil semua prestasi terkait profil
            $achievements = ApplicantAchievement::where('id_profile', $profile->id_profile)
                                                ->get();

            return response()->json([

                'data' => ApplicantAchievementResource::collection($achievements), // Asumsi resource ini sudah dibuat
            ], 200);

        } catch (\Exception $e) {
            // Mengembalikan 404 jika profile tidak ditemukan atau 500 jika error lain

                 return response()->json([
                    'success' => false,
                    'message' => 'Profil pendaftar belum lengkap.',
                ], 404);
            }
            \Log::error('Get Achievements Error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Gagal mengambil data prestasi',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

    /**
     * Add a new achievement for the applicant.
     * POST /api/registration/achievements
     */
    public function addAchievement(StoreAchievementRequest $request): JsonResponse
    {
        try {
            $user = Auth::user();
            $profile = ApplicantProfile::where('id_user', $user->id_user)->firstOrFail();

            // Ambil data yang divalidasi
            $data = $request->validated();
            $data['id_profile'] = $profile->id_profile;

            // 📌 LOGIKA UPLOAD FILE (PERBAIKAN)
            if ($request->hasFile('certificate_path')) {
                // Simpan file ke folder 'achievements' di storage public
                $path = $request->file('certificate_path')->store('achievements', 'public');
                
                // Timpa data 'certificate_path' dengan lokasi file di server
                $data['certificate_path'] = $path;
            }

            // Simpan ke database
            $achievement = ApplicantAchievement::create($data);

            return response()->json([
                'success' => true,
                'message' => 'Data prestasi berhasil ditambahkan',
                'data' => new ApplicantAchievementResource($achievement),
            ], 201);

        } catch (\Exception $e) {
            \Log::error('Add Achievement Error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Gagal menambahkan data prestasi',
                'error' => $e->getMessage(),
            ], 500);
        }
    }

    /**
     * Delete an achievement by ID.
     * DELETE /api/registration/achievements/{id}

    public function deleteAchievement($id): JsonResponse
    {
        try {
            $user = Auth::user();
            $profile = ApplicantProfile::where('id_user', $user->id_user)->firstOrFail();
            $achievement = ApplicantAchievement::findOrFail($id);

            // Cek otorisasi

                return response()->json([

            return response()->json([

                'message' => 'Prestasi berhasil dihapus',
            ], 200);
            
        } catch (\Exception $e) {
             if ($e instanceof \Illuminate\Database\Eloquent\ModelNotFoundException) {
                 return response()->json([
                    'success' => false,
                    'message' => 'Prestasi tidak ditemukan.',
                ], 404);
            }
            return response()->json([
                'success' => false,
                'message' => 'Gagal menghapus prestasi',
                'error' => $e->getMessage(),
            ], 500);
        }
    }
