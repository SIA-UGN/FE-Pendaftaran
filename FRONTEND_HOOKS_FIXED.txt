import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { registrationService } from "@/services/registrationService";
import toast from "react-hot-toast";

export const useAchievements = () => {
  const queryClient = useQueryClient();

  // Get achievements
  const {
    data: achievements,
    isLoading,
    error,
  } = useQuery({
    queryKey: ["achievements"],
    queryFn: async () => {
      const response = await registrationService.getAchievements();
      return response.data.data;
    },
  });

  // Add achievement
  const addAchievement = useMutation({
    mutationFn: (data) => registrationService.addAchievement(data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["achievements"] });
      queryClient.invalidateQueries({ queryKey: ["registrationProgress"] });
      toast.success("Prestasi berhasil ditambahkan");
    },
    onError: (error) => {
      toast.error(
        error.response?.data?.message || "Gagal menambahkan prestasi"
      );
    },
  });

  // Delete achievement
  const deleteAchievement = useMutation({
    mutationFn: (id) => registrationService.deleteAchievement(id),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["achievements"] });
      queryClient.invalidateQueries({ queryKey: ["registrationProgress"] });
      toast.success("Prestasi berhasil dihapus");
    },
    onError: (error) => {
      toast.error(error.response?.data?.message || "Gagal menghapus prestasi");
    },
  });

  return {
    achievements: achievements || [],
    isLoading,
    error,
    addAchievement: addAchievement.mutate,
    deleteAchievement: deleteAchievement.mutate,
    isAdding: addAchievement.isPending,
    isDeleting: deleteAchievement.isPending,
  };
};

import { useMutation, useQueryClient } from "@tanstack/react-query";
import { authService } from "@/services/authService";
import { useAuth as useAuthContext } from "@/contexts/AuthContext";
import { useRouter } from "next/navigation";
import toast from "react-hot-toast";
import { useEffect, useRef } from "react";

export const useLogin = () => {
  const { login: setAuth } = useAuthContext();
  const router = useRouter();
  const timeoutRef = useRef(null);

  // Cleanup timeout on unmount
  useEffect(() => {
    return () => {
      if (timeoutRef.current) {
        clearTimeout(timeoutRef.current);
      }
    };
  }, []);

  return useMutation({
    mutationFn: (credentials) => authService.login(credentials),
    onSuccess: (response) => {
      const { user, access_token } = response.data;

      // Set auth context
      setAuth(user, access_token);

      toast.success("Login berhasil");

      // Redirect based on role
      timeoutRef.current = setTimeout(() => {
        if (user.role === "admin") {
          router.push("/admin/dashboard");
        } else if (user.role === "manager") {
          router.push("/manager/dashboard");
        } else {
          router.push("/student/dashboard");
        }
      }, 500);
    },
    onError: (error) => {
      toast.error(error.response?.data?.message || "Login gagal");
    },
  });
};

---------- FILE: useDebounce.js ----------
import { useState, useEffect } from "react";

/**
 * Hook untuk debounce value
 * @param {any} value - Value yang akan di-debounce
 * @param {number} delay - Delay dalam milliseconds (default: 500ms)
 * @returns {any} Debounced value
 */
export const useDebounce = (value, delay = 500) => {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
};


---------- FILE: useFileUpload.js ----------
import { useState } from "react";
import toast from "react-hot-toast";

/**
 * Hook untuk handle file upload dengan validasi
 * @param {Object} options - Configuration options
 * @param {number} options.maxSize - Maximum file size in bytes (default: 2MB)
 * @param {string[]} options.allowedTypes - Allowed MIME types
 * @param {Function} options.onUpload - Upload handler function
 * @returns {Object} Upload utilities
 */
export const useFileUpload = ({
  maxSize = 2 * 1024 * 1024, // 2MB default
  allowedTypes = ["image/jpeg", "image/png", "image/jpg", "application/pdf"],
  onUpload,
} = {}) => {
  const [isUploading, setIsUploading] = useState(false);
  const [progress, setProgress] = useState(0);

  const validateFile = (file) => {
    if (!file) {
      toast.error("Tidak ada file yang dipilih");
      return false;
    }

    if (file.size > maxSize) {
      toast.error(`Ukuran file maksimal ${maxSize / (1024 * 1024)}MB`);
      return false;
    }

    if (!allowedTypes.includes(file.type)) {
      toast.error("Tipe file tidak didukung");
      return false;
    }

    return true;
  };

  const uploadFile = async (file, additionalData = {}) => {
    if (!validateFile(file)) return null;

    setIsUploading(true);
    setProgress(0);

    try {
      const formData = new FormData();
      formData.append("file", file);

      // Add additional data to formData
      Object.entries(additionalData).forEach(([key, value]) => {
        formData.append(key, value);
      });

      // Simulate progress (since we don't have real progress from axios)
      const progressInterval = setInterval(() => {
        setProgress((prev) => {
          if (prev >= 90) {
            clearInterval(progressInterval);
            return 90;
          }
          return prev + 10;
        });
      }, 200);

      const result = await onUpload(formData);

      clearInterval(progressInterval);
      setProgress(100);

      return result;
    } catch (error) {
      toast.error(error.response?.data?.message || "Gagal mengupload file");
      return null;
    } finally {
      setIsUploading(false);
      setTimeout(() => setProgress(0), 1000);
    }
  };

  return {
    uploadFile,
    isUploading,
    progress,
    validateFile,
  };
};

export const useUploadPaymentProof = () => {
  const queryClient = useQueryClient();
  const [uploadProgress, setUploadProgress] = useState(0);

  return useMutation({
    mutationFn: ({ paymentId, formData }) => {
      setUploadProgress(0);
      
      // Simulate progress
      const progressInterval = setInterval(() => {
        setUploadProgress((prev) => {
          if (prev >= 90) {
            clearInterval(progressInterval);
            return 90;
          }
          return prev + 10;
        });
      }, 300);

      return paymentService.uploadProof(paymentId, formData).finally(() => {
        clearInterval(progressInterval);
        setUploadProgress(100);
      });
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["myPayment"] });
      toast.success("Bukti pembayaran berhasil diupload");
      setTimeout(() => setUploadProgress(0), 1000);
    },
    onError: (error) => {
      toast.error(
        error.response?.data?.message || "Gagal mengupload bukti pembayaran"
      );
      setUploadProgress(0);
    },
    meta: {
      getProgress: () => uploadProgress,
    },
  });
};

import { useState } from "react";

/**
 * Hook untuk validasi step-by-step form
 * @param {number} totalSteps - Total jumlah step
 * @returns {Object} Step validation utilities
 */
export const useStepValidation = (totalSteps = 1) => {
  const [currentStep, setCurrentStep] = useState(1);
  const [completedSteps, setCompletedSteps] = useState([]);

  const goToNextStep = () => {
    if (currentStep < totalSteps) {
      setCurrentStep((prev) => prev + 1);
    }
  };

  const goToPreviousStep = () => {
    if (currentStep > 1) {
      setCurrentStep((prev) => prev - 1);
    }
  };

  const goToStep = (step) => {
    if (step >= 1 && step <= totalSteps) {
      setCurrentStep(step);
    }
  };

  const markStepCompleted = (step) => {
    if (!completedSteps.includes(step)) {
      setCompletedSteps((prev) => [...prev, step]);
    }
  };

  const markStepIncomplete = (step) => {
    setCompletedSteps((prev) => prev.filter((s) => s !== step));
  };

  const isStepCompleted = (step) => {
    return completedSteps.includes(step);
  };

  const isFirstStep = currentStep === 1;
  const isLastStep = currentStep === totalSteps;
  const canGoNext = currentStep < totalSteps;
  const canGoPrevious = currentStep > 1;

  const progressPercentage = (completedSteps.length / totalSteps) * 100;

  return {
    currentStep,
    completedSteps,
    goToNextStep,
    goToPreviousStep,
    goToStep,
    markStepCompleted,
    markStepIncomplete,
    isStepCompleted,
    isFirstStep,
    isLastStep,
    canGoNext,
    canGoPrevious,
    progressPercentage,
    totalSteps,
  };
};

/**
 * Hook untuk announcements dengan search dan pagination
 * @returns {Object} Announcements with search utilities
 */
export const useAnnouncementsWithSearch = () => {
  const [searchQuery, setSearchQuery] = useState("");
  const [currentPage, setCurrentPage] = useState(1);
  const [perPage] = useState(10);

  const debouncedSearch = useDebounce(searchQuery, 500);

  const {
    data: announcementsData,
    isLoading,
    error,
    refetch,
  } = useAnnouncements({
    search: debouncedSearch,
    page: currentPage,
    per_page: perPage,
  });

  // Reset to page 1 when search changes
  useEffect(() => {
    if (debouncedSearch) {
      setCurrentPage(1);
    }
  }, [debouncedSearch]);

  const handleSearch = (query) => {
    setSearchQuery(query);
  };

  const handlePageChange = (page) => {
    setCurrentPage(page);
  };

  return {
    announcements: announcementsData?.data || [],
    pagination: {
      current_page: announcementsData?.current_page || 1,
      last_page: announcementsData?.last_page || 1,
      per_page: announcementsData?.per_page || perPage,
      total: announcementsData?.total || 0,
    },
    searchQuery,
    isLoading,
    error,
    handleSearch,
    handlePageChange,
    refetch,
  };
