```javascript
  return useQuery({
    queryKey: ["payment", "my"],
    queryFn: paymentService.getMyPayment,
  });
};

// Service layer (paymentService.js)
export const paymentService = {
  getMyPayment: () => apiClient.get("/payment/my"),
};
```

2. CONTEXT → HOOK → COMPONENT PATTERN:
```
AuthContext (global state)
    ↓
useAuthContext() (hook)
    ↓
useLogin() (mutation hook)
    ↓
Login Page Component
```

3. MUTATION WITH INVALIDATION:
```javascript
export const useCreateManager = () => {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: adminService.createManager,
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["admin", "managers"] });
      toast.success("Manager created successfully!");
    },
  });
};
```

4. AUTO-REFETCH PATTERN:
