# lib/api.js
import axios from 'axios';
import toast from 'react-hot-toast';

const apiClient = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_URL || 'http://localhost:8000/api',
  headers: {
    'Content-Type': 'application/json',
  },
  timeout: 10000,
});

apiClient.interceptors.request.use((config) => {
  const token = localStorage.getItem('access_token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

apiClient.interceptors.response.use(
  (response) => response.data,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('access_token');
      localStorage.removeItem('user');
      
      if (typeof window !== 'undefined' && window.location.pathname !== '/login') {
        window.location.href = '/login';
      }
    }

    if (error.response?.status === 500) {
      toast.error('Server error. Silakan hubungi admin.');
    }

    if (!error.response) {
      toast.error('Koneksi terputus. Periksa internet Anda.');
    }

    return Promise.reject(error);
  }
);

export default apiClient;

# hooks/useAuth.js
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { authService } from '@/services/authService';
import toast from 'react-hot-toast';
import { useRouter } from 'next/navigation';
import { useEffect, useState } from 'react';
import { useAuthContext } from '@/contexts/AuthContext';

// Get auth user (disabled by default, use manually if needed)
export const useAuth = () => {
  const [token, setToken] = useState(null);
  
  useEffect(() => {
    setToken(localStorage.getItem('access_token'));
  }, []);
  
  return useQuery({
    queryKey: ['auth', 'user'],
    queryFn: authService.getUser,
    enabled: false, // Don't auto-fetch, prevent race conditions
    retry: false,
    staleTime: 5 * 60 * 1000,
  });
};

// Login mutation with localStorage and AuthContext integration
export const useLogin = () => {
  const router = useRouter();
  const queryClient = useQueryClient();
  const { updateUser } = useAuthContext(); // Get updateUser from context
  
  return useMutation({
    mutationFn: authService.login,
    onSuccess: async (response) => {
      // Response structure: response.data.data (axios unwraps one level)
      const data = response.data.data;
      
      if (!data || !data.user) {
        console.error('Invalid response structure:', response);
        toast.error('Login failed. Invalid response from server.');
        return;
      }
      
      // Store in localStorage
      localStorage.setItem('access_token', data.access_token);
      localStorage.setItem('user', JSON.stringify(data.user));
      
      // Update React Query cache
      queryClient.setQueryData(['auth', 'user'], data.user);
      
      // Update AuthContext state
      updateUser(data.user);
      
      toast.success('Login successful');
      
      // Role-based redirect
      const userRoles = data.user.roles || [];
      
      if (userRoles.includes('admin')) {
        router.push('/dashboard');
      } else if (userRoles.includes('manager')) {
        router.push('/manager');
      } else if (userRoles.includes('applicant') || userRoles.includes('student')) {
        router.push('/pendaftaran');
      } else {
        router.push('/');
      }
    },
    onError: (error) => {
      // Errors 401/500/network handled by interceptor
      // Only handle 4xx validation errors here
      if (!error.response || error.response?.status >= 500) return;
      
      const message = error.response?.data?.message || 
                     error.response?.data?.errors?.email?.[0] || 
                     'Login failed. Please try again.';
      toast.error(message);
    },
  });
};

// Register mutation with localStorage and AuthContext integration
export const useRegister = () => {
  const router = useRouter();
  const queryClient = useQueryClient();
  const { updateUser } = useAuthContext();
  
  return useMutation({
    mutationFn: authService.register,
    onSuccess: (response) => {
      const data = response.data.data;
      
      localStorage.setItem('access_token', data.access_token);
      localStorage.setItem('user', JSON.stringify(data.user));
      queryClient.setQueryData(['auth', 'user'], data.user);
      updateUser(data.user);
      toast.success('Registration successful');
      router.push('/pendaftaran');
    },
    onError: (error) => {
      if (!error.response || error.response?.status >= 500) return;
      
      const message = error.response?.data?.message || 
                     error.response?.data?.errors?.email?.[0] || 
                     'Registration failed. Please try again.';
      toast.error(message);
    },
  });
};

// Logout mutation with localStorage cleanup and AuthContext clearing
export const useLogout = () => {
  const router = useRouter();
  const queryClient = useQueryClient();
  const { clearUser } = useAuthContext();
  
  return useMutation({
    mutationFn: authService.logout,
    onSuccess: () => {
      localStorage.removeItem('access_token');
      localStorage.removeItem('user');
      queryClient.clear();
      clearUser(); // Clear AuthContext state
      toast.success('Logout successful');
      router.push('/');
    },
    onError: (error) => {
      localStorage.removeItem('access_token');
      localStorage.removeItem('user');
      queryClient.clear();
      clearUser(); // Clear AuthContext state even on error
      router.push('/login');
    },
  });
};

# hooks/useStepValidation.js
import { useEffect } from 'react';
import { useRouter } from 'next/navigation';
import { useRegistrationProgress } from './useRegistration';
import toast from 'react-hot-toast';

export const useStepValidation = (currentStep) => {
  const router = useRouter();
  const { data: progressData, isPending } = useRegistrationProgress();
  
  useEffect(() => {
    if (!isPending && progressData) {
      const { completed_steps = [], accessible_steps = [] } = progressData.data || {};
      
      if (!accessible_steps.includes(currentStep)) {
        const nextAccessibleStep = Math.max(...completed_steps, 0) + 1;
        
        toast.error(`Silakan selesaikan Step ${nextAccessibleStep} terlebih dahulu`);
        
        const stepRoutes = {
          1: '/pendaftaran/data-diri',
          2: '/pendaftaran/data-alamat',
          3: '/pendaftaran/data-akademik',
          4: '/pendaftaran/data-orangtua',
          5: '/pendaftaran/data-prestasi',
        };
        
        router.push(stepRoutes[nextAccessibleStep] || '/pendaftaran/data-diri');
      }
    }
  }, [progressData, isPending, currentStep, router]);
  
  return {
    isValidating: isPending,
    progressData: progressData?.data,
  };
};

export const useVerifyApplicant = () => {
  const queryClient = useQueryClient();

    mutationFn: ({ id, data }) => managerService.verifyApplicant(id, data),
    onSuccess: (data, { id }) => {
      queryClient.invalidateQueries({ queryKey: ['manager', 'applicants'] });
      queryClient.invalidateQueries({ queryKey: ['manager', 'applicant', id] });
      queryClient.invalidateQueries({ queryKey: ['manager', 'dashboard'] });

      const message = error?.response?.data?.message || 'Failed to update registration status';
      toast.error(message);
    },
  });

# contexts/AuthContext.js
import { createContext, useContext, useEffect, useState } from 'react';

const AuthContext = createContext({});

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  
  // Read user from localStorage on mount (no useAuth fetch to prevent race condition)
  useEffect(() => {
    const storedUser = localStorage.getItem('user');
    
    // Check for "undefined" or "null" strings (not actual null)
    if (storedUser && storedUser !== 'undefined' && storedUser !== 'null') {
      try {
        setUser(JSON.parse(storedUser));
      } catch (error) {
        console.error('Failed to parse user from localStorage:', error);
        localStorage.removeItem('user');
      }
    }
    setLoading(false);
  }, []);
  
  // Function to update user state (called by login/register mutations)
  const updateUser = (userData) => {
    setUser(userData);
  };
  
  // Function to clear user state (called by logout mutation)
  const clearUser = () => {
    setUser(null);
  };
  
  const hasRole = (role) => {
    return user?.roles?.includes(role) || false;
  };
  
  const hasPermission = (permission) => {
    return user?.permissions?.includes(permission) || false;
  };
  
  return (
    <AuthContext.Provider value={{
      user,
      loading,
      updateUser,      // Export updateUser function
      clearUser,       // Export clearUser function
      hasRole,
      hasPermission,
      isAuthenticated: !!user,
    }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuthContext = () => {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuthContext must be used within AuthProvider');
  }
  return context;
};

# middleware.js
import { NextResponse } from 'next/server';

export async function middleware(request) {
  const { pathname } = request.nextUrl;
  
  // Protected routes
  const protectedRoutes = ['/dashboard', '/registration', '/payment', '/admin'];
  const authRoutes = ['/login', '/register', '/forgot-password', '/reset-password'];
  
  // Check for token in cookies or authorization header
  const authHeader = request.headers.get('authorization');
  const token = authHeader?.replace('Bearer ', '') || 
                request.cookies.get('access_token')?.value;
  
  const isProtectedRoute = protectedRoutes.some(route => pathname.startsWith(route));
  const isAuthRoute = authRoutes.some(route => pathname.startsWith(route));
  
  // Redirect to login if accessing protected route without token
  if (isProtectedRoute && !token) {
    return NextResponse.redirect(new URL('/login', request.url));
  }
  
  // For authenticated users, check if they're trying to access auth pages
  if (isAuthRoute && token) {
    try {
      // Optional: Verify token by calling API
      const response = await fetch(`${process.env.NEXT_PUBLIC_API_URL}/auth/check`, {
        method: 'GET',
        headers: {
          'Cookie': request.headers.get('cookie') || '',
          'Authorization': token ? `Bearer ${token}` : '',
          'Accept': 'application/json',
        },
      });
      
      if (response.ok) {
        return NextResponse.redirect(new URL('/dashboard', request.url));
      }
    } catch (error) {
      // If verification fails, allow access to auth pages
      console.log('Token verification failed:', error);
    }
  }
  
  return NextResponse.next();
}

export const config = {
  matcher: [
    '/((?!api|_next/static|_next/image|favicon.ico).*)',
  ],
};

    1: { prev: null, next: '/pendaftaran/data-alamat' },
    2: { prev: '/pendaftaran/data-diri', next: '/pendaftaran/data-akademik' },
    3: { prev: '/pendaftaran/data-alamat', next: '/pendaftaran/data-orangtua' },

  };
  
  const route = stepRoutes[currentStep];

    <div className="flex justify-between pt-6">

  const formData = new FormData();

    // Sesuaikan dengan endpoint upload backend Anda
    const response = await fetch(`${process.env.NEXT_PUBLIC_API_URL}/upload`, {
      method: "POST",
      headers: {
        Authorization: `Bearer ${localStorage.getItem("access_token")}`,
      },
      body: formData,
    });
    
    if (!response.ok) throw new Error("Upload failed");
    
    const data = await response.json();
    return data.data.url; // Return URL dari backend
  } catch (error) {
    throw new Error("Failed to upload file");
  }

      method: "POST",
      headers: {
        Authorization: `Bearer ${localStorage.getItem("access_token")}`,
      },
      body: formData,
    });
    
    if (!response.ok) throw new Error("Upload failed");
    
    const data = await response.json();
    return data.data.url; // Return URL dari backend
  } catch (error) {
    throw new Error("Failed to upload file");
  }

// File upload service (Opsi A: Upload first, get URL, then submit)
const uploadFile = async (file) => {
  const formData = new FormData();
  formData.append("file", file);
  
  try {
    // Sesuaikan dengan endpoint upload backend Anda
    const response = await fetch(`${process.env.NEXT_PUBLIC_API_URL}/upload`, {
      method: "POST",
      headers: {
        Authorization: `Bearer ${localStorage.getItem("access_token")}`,
      },
      body: formData,
    });
    
    if (!response.ok) throw new Error("Upload failed");
    
    const data = await response.json();
    return data.data.url; // Return URL dari backend
  } catch (error) {
    throw new Error("Failed to upload file");
  }
};

// Validation Schema
const MAX_FILE_SIZE = 5000000; // 5MB
const ACCEPTED_FILE_TYPES = ["application/pdf", "image/jpeg", "image/jpg", "image/png"];

const achievementItemSchema = z.object({
  achievement_name: z.string().min(2, "Nama prestasi minimal 2 karakter"),
  year: z.string().min(4, "Tahun harus 4 digit").max(4, "Tahun harus 4 digit"),
  type: z.enum(["academic", "non_academic", "sports", "arts", "other"], {
    required_error: "Jenis prestasi harus dipilih",
  }),
  level: z.enum(["school", "district", "city", "province", "national", "international"], {
    required_error: "Tingkat prestasi harus dipilih",
  }),
  organizer: z.string().min(2, "Penyelenggara minimal 2 karakter"),
  rank: z.string().min(1, "Peringkat wajib diisi"),
  certificate_file: z.string().optional(), // URL file after upload
});

const achievementsSchema = z.object({
  skip_achievements: z.boolean().default(false),
  achievements: z.array(achievementItemSchema).optional(),
});

export default function DataPrestasiPage() {
  const { isValidating, progressData } = useStepValidation(5);
  const { data: registrationData } = useMyRegistration();
  const [isUploading, setIsUploading] = useState(false);
  const [skipAchievements, setSkipAchievements] = useState(false);
  const stepMutation = useRegistrationStep(5);

  const form = useForm({
    resolver: zodResolver(achievementsSchema),
    defaultValues: {
      skip_achievements: false,
      achievements: [
        {
          achievement_name: "",
          year: "",
          type: "",
          level: "",
          organizer: "",
          rank: "",
          certificate_file: "",
        },
      ],
    },
  });

  // Pre-fill form with existing data
  useEffect(() => {
    if (registrationData?.data?.registration?.achievements) {
      const achievements = registrationData.data.registration.achievements;
      if (achievements && achievements.length > 0) {
        form.reset({
          skip_achievements: false,
          achievements: achievements.map(achievement => ({
            achievement_name: achievement.achievement_name || "",
            year: achievement.year || "",
            type: achievement.type || "",
            level: achievement.level || "",
            organizer: achievement.organizer || "",
            rank: achievement.rank || "",
            certificate_file: achievement.certificate_file || "",
          })),
        });
        setSkipAchievements(false);
      }
    }
  }, [registrationData, form]);

  const { fields, append, remove } = useFieldArray({
    control: form.control,
    name: "achievements",
  });

  // Handle file upload for specific achievement
  const handleFileUpload = async (index, file) => {
    if (!file) return;

    // Validate file size
    if (file.size > MAX_FILE_SIZE) {
      toast.error("Ukuran file maksimal 5MB");
      return;
    }

    // Validate file type
    if (!ACCEPTED_FILE_TYPES.includes(file.type)) {
      toast.error("Format file harus PDF, JPG, atau PNG");
      return;
    }

    setIsUploading(true);
    try {
      const fileUrl = await uploadFile(file);
      form.setValue(`achievements.${index}.certificate_file`, fileUrl);
      toast.success(`Sertifikat berhasil diupload`);
    } catch (error) {
      toast.error("Gagal upload file, silakan coba lagi");
    } finally {
      setIsUploading(false);
    }
  };

  // Handle remove uploaded file
  const handleRemoveFile = (index) => {
    form.setValue(`achievements.${index}.certificate_file`, "");
  };

  const onSubmit = (data) => {
    // If skip achievements, send skip flag
    if (skipAchievements || data.skip_achievements) {
      stepMutation.mutate({ skip_achievements: true });
      return;
    }

    // Validate at least one achievement
    if (!data.achievements || data.achievements.length === 0) {
      toast.error("Tambahkan minimal 1 prestasi atau pilih 'Lewati Prestasi'");
      return;
    }

    // Validate all achievements have required fields
    const invalidAchievements = data.achievements.filter(
      (achievement) =>
        !achievement.achievement_name ||
        !achievement.year ||
        !achievement.type ||
        !achievement.level ||
        !achievement.organizer ||
        !achievement.rank
    );

    if (invalidAchievements.length > 0) {
      toast.error("Lengkapi semua field prestasi yang ditambahkan");
      return;
    }

    // Prepare payload
    const payload = {
      achievements: data.achievements.map((achievement) => ({
        achievement_name: achievement.achievement_name,
        year: parseInt(achievement.year),
        type: achievement.type,
        level: achievement.level,
        organizer: achievement.organizer,
        rank: achievement.rank,
        certificate_file: achievement.certificate_file || null,
      })),
    };

    // Data will be wrapped by service as { achievements: payload.achievements }
    stepMutation.mutate(payload);
  };

  if (isValidating) {
    return (
      <div className="flex items-center justify-center min-h-screen">
        <Loader2 className="w-8 h-8 animate-spin mr-2" />
        <p>Memvalidasi akses step...</p>
      </div>
    );
  }

  return (
    <div className="container mx-auto px-4 py-8 max-w-4xl">
      <ProgressIndicator />
      
      <Card>
        <CardHeader>
          <CardTitle>Informasi Prestasi (Opsional)</CardTitle>
        </CardHeader>
        <CardContent>
          <Form {...form}>
            <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-6">
              {/* Skip Achievements Option */}
              <div className="flex items-center space-x-2 p-4 bg-blue-50 border border-blue-200 rounded">
                <input
                  type="checkbox"
                  id="skip_achievements"
                  checked={skipAchievements}
                  onChange={(e) => {
                    setSkipAchievements(e.target.checked);
                    form.setValue("skip_achievements", e.target.checked);
                  }}
                  className="w-4 h-4"
                />
                <label htmlFor="skip_achievements" className="text-sm font-medium">
                  Saya tidak memiliki prestasi untuk dilaporkan (Lewati bagian ini)
                </label>
              </div>

              {/* Achievements Array */}
              {!skipAchievements && (
                <div className="space-y-6">
                  {fields.map((field, index) => (
                    <Card key={field.id} className="border-2">
                      <CardHeader className="flex flex-row items-center justify-between pb-4">
                        <CardTitle className="text-lg">
                          Prestasi #{index + 1}
                        </CardTitle>
                        {fields.length > 1 && (
                          <Button
                            type="button"
                            variant="destructive"
                            size="sm"
                            onClick={() => remove(index)}
                          >
                            <Trash2 className="w-4 h-4 mr-2" />
                            Hapus
                          </Button>
                        )}
                      </CardHeader>
                      <CardContent className="space-y-4">
                        {/* Achievement Name */}
                        <FormField
                          control={form.control}
                          name={`achievements.${index}.achievement_name`}
                          render={({ field }) => (
                            <FormItem>
                              <FormLabel>Nama Prestasi *</FormLabel>
                              <FormControl>
                                <Input
                                  placeholder="Contoh: Juara 1 Olimpiade Matematika"
                                  {...field}
                                />
                              </FormControl>
                              <FormMessage />
                            </FormItem>
                          )}
                        />

                        {/* Year, Type, Level */}
                        <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                          <FormField
                            control={form.control}
                            name={`achievements.${index}.year`}
                            render={({ field }) => (
                              <FormItem>
                                <FormLabel>Tahun *</FormLabel>
                                <FormControl>
                                  <Input
                                    placeholder="2024"
                                    maxLength={4}
                                    {...field}
                                  />
                                </FormControl>
                                <FormMessage />
                              </FormItem>
                            )}
                          />

                          <FormField
                            control={form.control}
                            name={`achievements.${index}.type`}
                            render={({ field }) => (
                              <FormItem>
                                <FormLabel>Jenis *</FormLabel>
                                <Select
                                  onValueChange={field.onChange}
                                  defaultValue={field.value}
                                >
                                  <FormControl>
                                    <SelectTrigger>
                                      <SelectValue placeholder="Pilih jenis" />
                                    </SelectTrigger>
                                  </FormControl>
                                  <SelectContent>
                                    <SelectItem value="academic">Akademik</SelectItem>
                                    <SelectItem value="non_academic">Non-Akademik</SelectItem>
                                    <SelectItem value="sports">Olahraga</SelectItem>
                                    <SelectItem value="arts">Seni</SelectItem>
                                    <SelectItem value="other">Lainnya</SelectItem>
                                  </SelectContent>
                                </Select>
                                <FormMessage />
                              </FormItem>
                            )}
                          />

                          <FormField
                            control={form.control}
                            name={`achievements.${index}.level`}
                            render={({ field }) => (
                              <FormItem>
                                <FormLabel>Tingkat *</FormLabel>
                                <Select
                                  onValueChange={field.onChange}
                                  defaultValue={field.value}
                                >
                                  <FormControl>
                                    <SelectTrigger>
                                      <SelectValue placeholder="Pilih tingkat" />
                                    </SelectTrigger>
                                  </FormControl>
                                  <SelectContent>
                                    <SelectItem value="school">Sekolah</SelectItem>
                                    <SelectItem value="district">Kecamatan</SelectItem>
                                    <SelectItem value="city">Kota/Kabupaten</SelectItem>
                                    <SelectItem value="province">Provinsi</SelectItem>
                                    <SelectItem value="national">Nasional</SelectItem>
                                    <SelectItem value="international">Internasional</SelectItem>
                                  </SelectContent>
                                </Select>
                                <FormMessage />
                              </FormItem>
                            )}
                          />
                        </div>

                        {/* Organizer & Rank */}
                        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                          <FormField
                            control={form.control}
                            name={`achievements.${index}.organizer`}
                            render={({ field }) => (
                              <FormItem>
                                <FormLabel>Penyelenggara *</FormLabel>
                                <FormControl>

                              </FormItem>

                                <FormLabel>Peringkat/Penghargaan *</FormLabel>
                                <FormControl>
                                  <Input
                                    placeholder="Contoh: Juara 1, Medali Emas"
                                    {...field}
                                  />
                                </FormControl>
                                <FormMessage />
                              </FormItem>

                          />
                        </div>

                        {/* Certificate Upload */}
                        <div className="space-y-2">
                          <Label>
                            Upload Sertifikat (PDF/JPG/PNG, Max 5MB)
                          </Label>

                                variant="ghost"
                                size="sm"

                              >

                              </Button>
                            </div>
                          ) : (
                            <div className="flex items-center gap-2">
                              <Input
                                type="file"
                                accept=".pdf,.jpg,.jpeg,.png"
                                onChange={(e) =>
                                  handleFileUpload(index, e.target.files?.[0])
                                }
                                disabled={isUploading}
                              />
                              {isUploading && (
                                <Loader2 className="w-4 h-4 animate-spin" />
                              )}
                            </div>
                          )}
                          <p className="text-xs text-gray-500">
                            Opsional - Upload jika ada sertifikat
                          </p>
                        </div>

                    </Card>
                  ))}

                  {/* Add Achievement Button */}
                  <Button
                    type="button"
                    variant="outline"
                    onClick={() =>
                      append({
                        achievement_name: "",
                        year: "",
                        type: "",
                        level: "",
                        organizer: "",
                        rank: "",
                        certificate_file: "",
                      })
                    }
                    className="w-full"
                  >
                    <Plus className="w-4 h-4 mr-2" />
                    Tambah Prestasi
                  </Button>
                </div>
              )}

              {/* Navigation using StepNavigation component */}
              <StepNavigation 
                currentStep={5} 
                isSubmitting={stepMutation.isPending || isUploading}
                showBack={true}
              />
            </form>
          </Form>
        </CardContent>
      </Card>
    </div>
  );
}

### D. Flow Validasi:

1. User akses halaman step (misal /pendaftaran/data-akademik yang adalah step 3)
2. useStepValidation(3) hook akan fetch /registration/progress
3. Hook cek apakah 3 ada di accessible_steps
4. Jika tidak ada, redirect ke step yang accessible (misal step 2 belum complete, redirect ke step 2)
5. Jika valid, render halaman dengan loading state hilang
6. useMyRegistration() fetch data existing untuk pre-fill form
7. useEffect update form.reset() dengan data dari database
8. User edit/lengkapi form
9. Submit → mutation → invalidate queries → auto redirect ke step berikutnya

StepNavigation:
  - currentStep: number (1-5)
  - isSubmitting: boolean
  - showBack: boolean (false untuk step 1)
  - prevRoute: string (optional, override default route)
